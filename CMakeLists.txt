cmake_minimum_required(VERSION 3.16)
project(AidLuxGridshot LANGUAGES CXX)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio dnn)
find_package(Threads REQUIRED)
option(GRIDSHOT_WITH_QNN "Build the external qnn_runtime integration" OFF)
set(QNN_RUNTIME_INCLUDE_DIR "" CACHE PATH "Directory that contains qnn_runtime/session.hpp")
set(QNN_RUNTIME_LIBRARY "" CACHE FILEPATH "Path to the locally built libqnn_runtime.a")

function(gridshot_target target src)
  add_executable(${target} ${src})
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(${target} SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(${target} PRIVATE ${OpenCV_LIBS} Threads::Threads)
  if(GRIDSHOT_WITH_QNN)
    if(NOT EXISTS "${QNN_RUNTIME_INCLUDE_DIR}/qnn_runtime/session.hpp" OR NOT EXISTS "${QNN_RUNTIME_LIBRARY}")
      message(FATAL_ERROR "Set QNN_RUNTIME_INCLUDE_DIR and QNN_RUNTIME_LIBRARY to your local adapter build")
    endif()
    target_compile_definitions(${target} PRIVATE GRIDSHOT_HAVE_QNN=1)
    target_include_directories(${target} PRIVATE "${QNN_RUNTIME_INCLUDE_DIR}")
    target_link_libraries(${target} PRIVATE "${QNN_RUNTIME_LIBRARY}" ${OpenCV_LIBS} ${CMAKE_DL_LIBS})
  endif()
endfunction()

gridshot_target(aim-live live/aim-live.cpp)
include(CTest)
if(BUILD_TESTING)
  gridshot_target(gridshot-check live/check.cpp)
  target_compile_options(gridshot-check PRIVATE -Wno-return-type)
  add_test(NAME feedback_recovery_and_mock_hid COMMAND gridshot-check "${CMAKE_CURRENT_BINARY_DIR}/mock-hid.bin")
endif()
